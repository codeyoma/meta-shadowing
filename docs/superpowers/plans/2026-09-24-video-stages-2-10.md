## Task 1: Define the selected-segment timeline

**Interfaces:**

```swift
struct VideoSegmentTimeline: Sendable {
  struct Segment: Sendable { let start: Double; let end: Double }
  struct Location: Sendable { let member: Int; let mediaSeconds: Double }
  let segments: [Segment]
  let duration: Double
  init(segments: [Segment]) throws
  func locate(_ position: Double) throws -> Location
  func position(member: Int, mediaSeconds: Double) -> Double
}
```

`duration` is the sum of selected durations. Require 1–4 ordered, finite, positive-length, nonoverlapping segments. `locate` rejects nonfinite/out-of-range positions; an interior boundary maps to the next member, while total duration maps to the final member's end. `position` clamps the current member offset and adds only earlier selected durations. Manifest indices are validated separately in Task 2.

- [ ] Add a failing Swift test using this concrete synthetic boundary case:

```swift
@Test func mapsSelectedTimeWithoutCountingGaps() throws {
  let timeline = try VideoSegmentTimeline(segments: [
    .init(start: 0.2, end: 0.6), .init(start: 1.8, end: 2.4)])
  #expect(abs(timeline.duration - 1) < 0.000001)
  let boundary = try timeline.locate(0.6 - 0.2)
  #expect(boundary.member == 1)
  #expect(abs(boundary.mediaSeconds - 1.8) < 0.000001)
  #expect(abs(timeline.position(member: 1, mediaSeconds: 2) - 0.6) < 0.000001)
  #expect(try timeline.locate(timeline.duration).member == 1)
  #expect(throws: (any Error).self) { try timeline.locate(.nan) }
  #expect(throws: (any Error).self) { try timeline.locate(1.1) }
}
```

- [ ] Generate the test project and run the native test command below. Confirm failure is the missing timeline implementation, not project setup.
- [ ] Implement the timeline using a bounded loop. The location selection rule is:

```swift
var prefix = 0.0
for (index, segment) in segments.enumerated() {
  let length = segment.end - segment.start
  if position < prefix + length || index == segments.count - 1 {
    return Location(member: index, mediaSeconds: segment.start + position - prefix)
  }
  prefix += length
}
throw VideoError.invalid
```
